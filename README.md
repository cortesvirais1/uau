        // Contagem regressiva para a oferta limitada (expira em 24 horas)
        const countdownElement = document.getElementById('countdown');
        const targetDate = new Date().getTime() + (24 * 60 * 60 * 1000); // 24 horas a partir de agora

        function updateCountdown() {
            const now = new Date().getTime();
            const distance = targetDate - now;

            if (distance > 0) {
                const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
                const seconds = Math.floor((distance % (1000 * 60)) / 1000);
                countdownElement.innerHTML = hours + "h " + minutes + "m " + seconds + "s";
            } else {
                countdownElement.innerHTML = "Expirada";
                document.querySelector('.limited-offer').style.opacity = '0.5';
            }
        }

        setInterval(updateCountdown, 1000);
        updateCountdown(); // Inicializa imediatamente
